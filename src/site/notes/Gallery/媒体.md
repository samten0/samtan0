async function loadData() {
    // 显示加载指示器
    const loadingIndicator = dv.el("div", "加载中...", { cls: "loading-indicator" });
    container.appendChild(loadingIndicator);
    
    // 获取所有页面
    const pages = dv.pages('"Gallery/gallery_source/02-电影阅读"')
        .filter(p => p.file)
        .map(p => {
            // 处理封面图片
            let cover = "";
            if (p.image) cover = p.image;
            else if (p.cover) cover = p.cover;
            else if (p.poster) cover = p.poster;
            else if (p.thumbnail) cover = p.thumbnail;
            else if (p.coverImage) cover = p.coverImage;
            
            // 处理创作者信息
            let creator = "";
            if (p.author) creator = p.author;
            else if (p.director) creator = p.director;
            else if (p.artist) creator = p.artist;
            else if (p.developer) creator = p.developer;
            
            // 处理日期信息
            let date = "";
            if (p.datePublished) date = p.datePublished;
            else if (p.releaseDate) date = p.releaseDate;
            else if (p.year) date = p.year;
            else if (p.publishYear) date = p.publishYear;
            
            // 处理评分
            let score = typeof p.score === 'number' ? p.score : parseFloat(p.score || 0);
            
            // 处理类型
            let type = p.type || "";
            
            // 处理状态
            let status = p.status !== undefined ? p.status : false;
            
            // 处理标签
            let tags = p.tags || [];
            if (!Array.isArray(tags)) {
                tags = [tags];
            }
            
            return {
                title: p.title || p.file.name,
                creator: creator,
                date: date,
                year: p.year || p.publishYear || "",
                score: score,
                type: type,
                status: status,
                cover: cover,
                path: p.file.path,
                file: p.file,
                // 额外信息
                genre: p.genre || "",
                publisher: p.publisher || "",
                platform: p.platform || "",
                tags: tags,
                // 原始数据，用于全文搜索
                rawContent: p.file.content || ""
            };
        });
    
    // 移除加载指示器
    container.removeChild(loadingIndicator);
    return pages;
}

// 保存筛选器状态到本地存储
function saveFiltersToStorage(filters) {
    try {
        localStorage.setItem('mediaLibraryFilters', JSON.stringify(filters));
        localStorage.setItem('mediaLibraryFiltersTimestamp', Date.now().toString());
    } catch (e) {
        console.error('保存筛选器状态失败:', e);
    }
}

// 从本地存储加载筛选器状态
function loadFiltersFromStorage() {
    try {
        const filtersJson = localStorage.getItem('mediaLibraryFilters');
        const timestamp = parseInt(localStorage.getItem('mediaLibraryFiltersTimestamp') || '0');
        
        // 检查是否存在筛选器状态且未过期（30分钟内有效）
        if (filtersJson && Date.now() - timestamp < 30 * 60 * 1000) {
            return JSON.parse(filtersJson);
        }
    } catch (e) {
        console.error('加载筛选器状态失败:', e);
    }
    return null;
}

// 创建分页
function createPagination(totalPages, currentPage, onPageChange) {
    const pagination = dv.el("div", "", { cls: "pagination" });
    
    // 上一页按钮
    if (currentPage > 1) {
        const prevBtn = dv.el("div", "上一页", { cls: "page-btn" });
        prevBtn.addEventListener('click', () => onPageChange(currentPage - 1));
        pagination.appendChild(prevBtn);
    }
    
    // 页码按钮
    for (let i = 1; i <= totalPages; i++) {
        if (
            i === 1 || 
            i === totalPages || 
            (i >= currentPage - 2 && i <= currentPage + 2)
        ) {
            const pageBtn = dv.el("div", i.toString(), { 
                cls: `page-btn ${i === currentPage ? 'active' : ''}`
            });
            pageBtn.addEventListener('click', () => onPageChange(i));
            pagination.appendChild(pageBtn);
        } else if (
            (i === currentPage - 3 || i === currentPage + 3) &&
            pagination.lastChild && 
            !pagination.lastChild.classList.contains('page-btn-ellipsis')
        ) {
            const ellipsis = dv.el("div", "...", { cls: "page-btn-ellipsis" });
            pagination.appendChild(ellipsis);
        }
    }
    
    // 下一页按钮
    if (currentPage < totalPages) {
        const nextBtn = dv.el("div", "下一页", { cls: "page-btn" });
        nextBtn.addEventListener('click', () => onPageChange(currentPage + 1));
        pagination.appendChild(nextBtn);
    }
    
    return pagination;
}

// 获取封面图片URL
function getCoverImageUrl(page) {
    if (!page || !page.cover) return '';
    
    const coverPath = page.cover;
    
    if (coverPath.startsWith('http') || coverPath.startsWith('data:')) {
        return coverPath;
    } else if (coverPath.includes('[[') || coverPath.includes('![[')) {
        const cleanPath = coverPath.replace(/[!\[\]]/g, '').split('|')[0].trim();
        try {
            return app.vault.adapter.getResourcePath(cleanPath);
        } catch (e) {
            return '';
        }
    } else {
        try {
            return app.vault.adapter.getResourcePath(coverPath);
        } catch (e) {
            return coverPath;
        }
    }
}

// 格式化日期为年月日格式
function formatDate(dateStr) {
    try {
        if (!dateStr) return '';
        
        // 如果只是年份
        if (/^\d{4}$/.test(dateStr)) {
            return dateStr;
        }
        
        const date = new Date(dateStr);
        if (isNaN(date.getTime())) {
            return dateStr;
        }
        
        // 返回年月日格式 (YYYY-MM-DD)
        const year = date.getFullYear();
        const month = (date.getMonth() + 1).toString().padStart(2, '0');
        const day = date.getDate().toString().padStart(2, '0');
        return `${year}-${month}-${day}`;
    } catch (e) {
        return dateStr || '';
    }
}

    notification.textContent = message;
    notification.style.position = 'fixed';
    notification.style.bottom = '20px';
    notification.style.right = '20px';
    notification.style.padding = '10px 20px';
    notification.style.backgroundColor = type === 'success' ? '#4CAF50' : '#F44336';
    notification.style.color = 'white';
    notification.style.borderRadius = '4px';
    notification.style.zIndex = '1000';
    notification.style.opacity = '0';
    notification.style.transition = 'opacity 0.3s ease';
    
    document.body.appendChild(notification);
    
    // 淡入
    setTimeout(() => {
        notification.style.opacity = '1';
    }, 10);
    
    // 3秒后淡出
    setTimeout(() => {
        notification.style.opacity = '0';
        setTimeout(() => {
            document.body.removeChild(notification);
        }, 300);
    }, 3000);
}

// 更新笔记状态
async function updateNoteStatus(page, newStatus) {
    try {
        // 获取文件内容
        const content = await app.vault.read(app.vault.getAbstractFileByPath(page.path));
        
        // 检查是否已有status字段
        const hasStatus = /status:\s*(true|false)/i.test(content);
        
        let newContent;
        if (hasStatus) {
            // 替换现有status
            newContent = content.replace(/status:\s*(true|false)/i, `status: ${newStatus}`);
        } else {
            // 在frontmatter中添加status字段
            if (content.startsWith('---')) {
                const endOfFrontmatter = content.indexOf('---', 3);
                if (endOfFrontmatter !== -1) {
                    newContent = content.slice(0, endOfFrontmatter) + `status: ${newStatus}\n` + content.slice(endOfFrontmatter);
                } else {
                    // 无法找到frontmatter结束标记，添加到开头
                    newContent = `---\nstatus: ${newStatus}\n---\n${content.slice(3)}`;
                }
            } else {
                // 没有frontmatter，创建一个
                newContent = `---\nstatus: ${newStatus}\n---\n\n${content}`;
            }
        }
        
        // 写入文件
        await app.vault.modify(app.vault.getAbstractFileByPath(page.path), newContent);
        return true;
    } catch (e) {
        console.error('更新笔记状态失败:', e);
        return false;
    }
}

// 保存当前页码到本地存储
function saveCurrentPageToStorage(page) {
    try {
        localStorage.setItem('mediaLibraryCurrentPage', page.toString());
        localStorage.setItem('mediaLibraryPageTimestamp', Date.now().toString());
    } catch (e) {
        console.error('保存当前页码失败:', e);
    }
}

// 从本地存储加载当前页码
function loadCurrentPageFromStorage() {
    try {
        const pageStr = localStorage.getItem('mediaLibraryCurrentPage');
        const timestamp = parseInt(localStorage.getItem('mediaLibraryPageTimestamp') || '0');
        
        // 检查是否存在页码且未过期（30分钟内有效）
        if (pageStr && Date.now() - timestamp < 30 * 60 * 1000) {
            return parseInt(pageStr);
        }
    } catch (e) {
        console.error('加载当前页码失败:', e);
    }
    return 1; // 默认返回第一页
}

// 渲染画廊
function renderGallery(pages, filters, currentPage, onPageChange) {
    // 筛选数据
    let filteredPages = [...pages];
    
    // 搜索筛选 - 改进搜索算法，支持多字段和全文搜索
    if (filters.search) {
        const searchTerms = filters.search.toLowerCase().split(/\s+/).filter(term => term.length > 0);
        
        if (searchTerms.length > 0) {
            filteredPages = filteredPages.filter(page => {
                // 创建一个包含所有可搜索字段的文本
                const searchableText = [
                    page.title,
                    page.creator,
                    page.type,
                    page.genre,
                    page.publisher,
                    page.platform,
                    page.year?.toString(),
                    page.date?.toString(),
                    // 添加标签作为搜索字段
                    ...(page.tags || []),
                    // 添加原始内容用于全文搜索
                    page.rawContent
                ].filter(Boolean).join(' ').toLowerCase();
                
                // 所有搜索词都必须匹配
                return searchTerms.every(term => searchableText.includes(term));
            });
        }
    }
    
    // 类型筛选
    if (filters.type && filters.type !== 'all') {
        filteredPages = filteredPages.filter(page => page.type === filters.type);
    }
    
    // 评分筛选
    if (filters.rating && filters.rating !== 'all') {
        const targetRating = parseInt(filters.rating);
        filteredPages = filteredPages.filter(page => {
            const score = typeof page.score === 'number' ? page.score : parseFloat(page.score || 0);
            if (targetRating === 5) return score >= 9;
            if (targetRating === 4) return score >= 7 && score < 9;
            if (targetRating === 3) return score >= 5 && score < 7;
            if (targetRating === 2) return score >= 3 && score < 5;
            if (targetRating === 1) return score > 0 && score < 3;
            return false;
        });
    }
    
    // 状态筛选
    if (filters.status && filters.status !== 'all') {
        const targetStatus = filters.status === 'true';
        filteredPages = filteredPages.filter(page => page.status === targetStatus);
    }
    
    // 排序
    if (filters.sort) {
        filteredPages = filteredPages.sort((a, b) => {
            let valueA, valueB;
            
            if (filters.sort.field === 'score') {
                valueA = typeof a.score === 'number' ? a.score : parseFloat(a.score || 0);
                valueB = typeof b.score === 'number' ? b.score : parseFloat(b.score || 0);
            } else if (filters.sort.field === 'title') {
                valueA = (a.title || '').toLowerCase();
                valueB = (b.title || '').toLowerCase();
            } else if (filters.sort.field === 'date') {
                valueA = a.date ? new Date(a.date) : new Date(0);
                valueB = b.date ? new Date(b.date) : new Date(0);
            } else {
                valueA = a[filters.sort.field] || '';
                valueB = b[filters.sort.field] || '';
            }
            
            if (filters.sort.direction === 'asc') {
                return valueA > valueB ? 1 : -1;
            } else {
                return valueA < valueB ? 1 : -1;
            }
        });
    }
    
    // 计算分页
    const totalPages = Math.max(1, Math.ceil(filteredPages.length / ITEMS_PER_PAGE));
    const validCurrentPage = Math.min(totalPages, Math.max(1, currentPage));
    const startIndex = (validCurrentPage - 1) * ITEMS_PER_PAGE;
    const endIndex = startIndex + ITEMS_PER_PAGE;
    const paginatedPages = filteredPages.slice(startIndex, endIndex);
    
    // 保存当前页码到本地存储
    saveCurrentPageToStorage(validCurrentPage);
    
    // 清空现有内容
    const galleryGrid = document.querySelector('.gallery-grid');
    galleryGrid.innerHTML = '';
    
    // 显示无结果消息
    if (filteredPages.length === 0) {
        galleryGrid.innerHTML = '<div class="no-results">没有找到匹配的结果</div>';
        return { totalPages, currentPage: 1 };
    }
    
    // 渲染卡片
    paginatedPages.forEach(page => {
        const card = dv.el("div", "", { cls: "gallery-card" });
        
        // 创建图片元素
        const img = document.createElement("img");
        img.src = page.cover ? getCoverImageUrl(page) : getDefaultCover(page.type);
        img.alt = page.title;
        img.onerror = () => {
            img.src = getDefaultCover(page.type);
        };
        card.appendChild(img);
        
        // 添加类型标签
        if (page.type) {
            const typeBadge = dv.el("div", getTypeIcon(page.type) + " " + page.type, { cls: "type-badge" });
            card.appendChild(typeBadge);
        }
        
        // 添加评分标签 - 固定在右下角
        if (page.score) {
            const ratingBadge = dv.el("div", getRatingStars(page.score), { cls: "rating-badge" });
            card.appendChild(ratingBadge);
        }
        
        // 添加日期标签 - 固定在左下角，显示年月日格式
        if (page.date || page.year) {
            const dateText = formatDate(page.date || page.year);
            if (dateText) {
                const dateBadge = dv.el("div", dateText, { cls: "date-badge" });
                card.appendChild(dateBadge);
            }
        }
        
        // 添加状态标签
        const statusText = page.status ? "已完成" : "未完成";
        const statusBadge = dv.el("div", statusText, { cls: "status-badge" });
        statusBadge.addEventListener('click', async (e) => {
            e.stopPropagation(); // 阻止事件冒泡
            
            const newStatus = !page.status;
            const success = await updateNoteStatus(page, newStatus);
            
            if (success) {
                page.status = newStatus;
                statusBadge.textContent = newStatus ? "已完成" : "未完成";
                showNotification(`已将 "${page.title}" 标记为 ${newStatus ? "已完成" : "未完成"}`);
            } else {
                showNotification("更新状态失败，请重试", "error");
            }
        });
        card.appendChild(statusBadge);
        
        // 添加信息层
        const info = dv.el("div", "", { cls: "card-info" });
        
        // 标题 - 使用渐变紫色
        const title = dv.el("div", page.title, { cls: "card-title" });
        info.appendChild(title);
        
        // 创作者
        if (page.creator) {
            const creator = dv.el("div", page.creator, { cls: "card-creator" });
            creator.style.fontSize = "0.9em";
            creator.style.marginBottom = "5px";
            info.appendChild(creator);
        }
        
        // 添加额外信息
        if (page.genre) {
            const genre = dv.el("div", `类型: ${page.genre}`, { cls: "card-genre" });
            genre.style.fontSize = "0.8em";
            genre.style.opacity = "0.8";
            info.appendChild(genre);
        }
        
        if (page.publisher) {
            const publisher = dv.el("div", `出版: ${page.publisher}`, { cls: "card-publisher" });
            publisher.style.fontSize = "0.8em";
            publisher.style.opacity = "0.8";
            info.appendChild(publisher);
        }
        
        if (page.platform) {
            const platform = dv.el("div", `平台: ${page.platform}`, { cls: "card-platform" });
            platform.style.fontSize = "0.8em";
            platform.style.opacity = "0.8";
            info.appendChild(platform);
        }
        
        // 在信息层中也显示评分和日期，但格式更详细
        if (page.score) {
            const scoreInfo = dv.el("div", `评分: ${page.score}`, { cls: "card-score" });
            scoreInfo.style.fontSize = "0.8em";
            scoreInfo.style.opacity = "0.8";
            info.appendChild(scoreInfo);
        }
        
        if (page.date || page.year) {
            const dateText = formatDate(page.date || page.year);
            if (dateText) {
                const dateInfo = dv.el("div", `日期: ${dateText}`, { cls: "card-date" });
                dateInfo.style.fontSize = "0.8em";
                dateInfo.style.opacity = "0.8";
                info.appendChild(dateInfo);
            }
        }
        
        card.appendChild(info);
        
        // 添加点击事件
        card.addEventListener('click', () => {
            app.workspace.openLinkText(page.path, '', false);
        });
        
        galleryGrid.appendChild(card);
    });
    
    // 更新分页
    const paginationContainer = document.querySelector('.pagination');
    if (paginationContainer) {
        container.removeChild(paginationContainer);
    }
    
    if (totalPages > 1) {
        const pagination = createPagination(totalPages, validCurrentPage, onPageChange);
        container.appendChild(pagination);
    }
    
    return { totalPages, currentPage: validCurrentPage };
}

// 初始化画廊
async function initGallery() {
    // 创建画廊网格
    const galleryGrid = createGalleryGrid();
    container.appendChild(galleryGrid);
    
    // 加载数据
    const pages = await loadData();
    
    // 从本地存储加载筛选器状态
    const savedFilters = loadFiltersFromStorage() || {
        search: '',
        type: 'all',
        rating: 'all',
        status: 'all',
        sort: {field: 'score', direction: 'desc'}
    };
    
    // 从本地存储加载当前页码
    let currentPage = loadCurrentPageFromStorage();
    
    // 页面变更处理函数
    const handlePageChange = (page) => {
        currentPage = page;
        saveCurrentPageToStorage(page);
        renderGallery(pages, savedFilters, currentPage, handlePageChange);
    };
    
    // 创建筛选器
    const filterBar = createHorizontalFilterBar(pages, (newFilters) => {
        // 更新筛选器状态
        Object.assign(savedFilters, newFilters);
        
        // 重置到第一页
        currentPage = 1; 
        saveCurrentPageToStorage(currentPage);
        
        // 重新渲染
        const { totalPages, currentPage: validPage } = renderGallery(pages, savedFilters, currentPage, handlePageChange);
        currentPage = validPage;
    });
    
    // 添加筛选器到容器
    container.insertBefore(filterBar, galleryGrid);
    
    // 初始渲染
    const { totalPages, currentPage: validPage } = renderGallery(pages, savedFilters, currentPage, handlePageChange);
    currentPage = validPage;
}

// 添加容器到页面
dv.container.appendChild(container);

// 初始化画廊
initGallery();
